               "\nEmail: " + email +

}


// ================= BILLING =================
class Billing {
    private int billID;
    private Date billDate;
    private double totalAmount;
    private double discount;       // FIXED: added discount field
    private double tax;
    private double grandTotal;

    private List<BillDetail> items = new ArrayList<>();

    public Billing(int billID, Date billDate,
                   double discount, double tax) {
        this.billID = billID;
        this.billDate = billDate;
        this.discount = discount;
        this.tax = tax;
    }

    public void addItem(BillDetail item) {
        items.add(item);
        System.out.println("Product added to bill.");
    }

    public void removeItem(BillDetail item) {
        items.remove(item);
        System.out.println("Product removed from bill.");
    }

    public double calculateTotal() {
        totalAmount = 0;

        for (BillDetail item : items) {
            totalAmount += item.calculateSubtotal();
        }

        grandTotal = totalAmount - discount + tax;

        return grandTotal;
    }

    public void generateBill() {

        // Calculate total before displaying the bill
        calculateTotal();

        System.out.println("\n========== BILL ==========");
        System.out.println("Bill ID: " + billID);
        System.out.println("Bill Date: " + billDate);

        for (BillDetail item : items) {
            System.out.println(item.getDetails());
        }

        System.out.println("--------------------------");
        System.out.println("Total Amount: " + totalAmount);
        System.out.println("Discount: " + discount);
        System.out.println("Tax: " + tax);
        System.out.println("Grand Total: " + grandTotal);
        System.out.println("==========================");
    }
}


// ================= BILL DETAIL =================
class BillDetail {
    private int billID;
    private int productID;
    private int quantity;
    private double unitPrice;

    public BillDetail(int billID, int productID,
                      int quantity, double unitPrice) {
        this.billID = billID;
        this.productID = productID;
        this.quantity = quantity;
        this.unitPrice = unitPrice;
    }

    public double calculateSubtotal() {
        return quantity * unitPrice;
    }

    public String getDetails() {
        return "Product ID: " + productID +
               " | Quantity: " + quantity +
               " | Unit Price: " + unitPrice +
               " | Subtotal: " + calculateSubtotal();
    }
}


// ================= PAYMENT =================
class Payment {
    private int paymentID;
    private int billID;
    private String paymentMode;
    private double amount;
    private Date paymentDate;
    private String transactionID;

    public Payment(int paymentID, int billID,
                   String paymentMode, double amount,
                   Date paymentDate, String transactionID) {
        this.paymentID = paymentID;
        this.billID = billID;
        this.paymentMode = paymentMode;
        this.amount = amount;
        this.paymentDate = paymentDate;
        this.transactionID = transactionID;
    }

    public void processPayment() {
        System.out.println("\nPayment processed successfully.");
        System.out.println("Payment Mode: " + paymentMode);
        System.out.println("Amount: " + amount);
        System.out.println("Payment Date: " + paymentDate);
    }

    public String getPaymentDetails() {
        return "Payment ID: " + paymentID +
               "\nBill ID: " + billID +
               "\nPayment Mode: " + paymentMode +
               "\nAmount: " + amount +
               "\nPayment Date: " + paymentDate +
               "\nTransaction ID: " + transactionID;
    }

    public void refund() {
        System.out.println("Payment refunded.");
    }
}

// ================= INVENTORY =================
class Inventory {
    private int inventoryID;
    private int productID;
    private int stockQuantity;
    private int reorderLevel;
    private Date lastUpdated;

    public Inventory(int inventoryID, int productID,
                     int stockQuantity, int reorderLevel) {
        this.inventoryID = inventoryID;
        this.productID = productID;
        this.stockQuantity = stockQuantity;
        this.reorderLevel = reorderLevel;
        this.lastUpdated = new Date();
    }

    public void updateStock(int quantity) {
        stockQuantity += quantity;
        lastUpdated = new Date();

        System.out.println("Stock updated.");
    }

    public int checkStock() {
        return stockQuantity;
    }

    public String getInventoryDetails() {
        return "Inventory ID: " + inventoryID +
               "\nProduct ID: " + productID +
               "\nStock Quantity: " + stockQuantity +
               "\nReorder Level: " + reorderLevel +
               "\nLast Updated: " + lastUpdated;
    }
}
